Como integrar no seu site

Adicione 3 linhas ao HTML do seu site. Pronto.

<!-- 1. Configure o widget -->
<script>
  window.NaroscoChatConfig = {
    webhookUrl: 'https://seu-n8n.com/webhook/clinic-chat',
    clinicName: 'Clínica São Lucas',
    agentName:  'Ana',
    agentEmoji: '👩‍⚕️',

<!-- 2. CSS -->
<link rel="stylesheet" href="chat.css">

<!-- 3. JS (antes do </body>) -->
<script src="chat.js"></script>
